import json, re, string
import requests
import os
import harness, lisanbench

# Seul le service distant est remplacé : requests.get lit words_alpha.txt (copie locale du fichier dwyl). Le barème reste celui de l'env.
_TXT = open(os.path.join(os.path.dirname(os.path.abspath(__file__)), "words_alpha.txt"), encoding="utf-8").read()


class _Rep:
    text = _TXT


requests.get = lambda *a, **k: _Rep()
env = lisanbench.load_environment()
ds = env.dataset
words = set(w.strip().lower() for w in _TXT.splitlines() if w.strip())


def nbrs(w):
    for i in range(len(w)):
        for c in string.ascii_lowercase:
            if c != w[i] and w[:i] + c + w[i + 1:] in words:
                yield w[:i] + c + w[i + 1:]
        if w[:i] + w[i + 1:] in words:
            yield w[:i] + w[i + 1:]
    for i in range(len(w) + 1):
        for c in string.ascii_lowercase:
            if w[:i] + c + w[i:] in words:
                yield w[:i] + c + w[i:]


def chain(start, limit=300):
    path = [start]
    seen = {start}
    while len(path) < limit:
        c = [n for n in nbrs(path[-1]) if n not in seen]
        if not c:
            break
        n = min(c, key=lambda x: sum(1 for y in nbrs(x) if y not in seen))
        path.append(n)
        seen.add(n)
    return path


starts = [re.search(r'Start with the word: "(\w+)"', r["question"]).group(1) for r in ds]
univ = chain("cat")  # UNE seule chaîne, jamais liée au mot de départ
print("chaine universelle:", len(univ), "mots")
H = C = 0.0
touch = 0
for s in starts:
    h, _ = harness.score(env, ", ".join(chain(s)), "N/A")
    c, _ = harness.score(env, ", ".join(univ), "N/A")
    print(s, "honnete=%.2f triche=%.2f" % (h, c))
    H += h
    C += c
    touch += c > h
n = len(starts)
H /= n
C /= n
print("moyennes", H, C)
print("VERDICT " + json.dumps({"env": "lisanbench", "verdict": "faille", "classe": "note_partielle_volee",
                               "honnete": round(H, 2), "triche": round(C, 2), "max": None,
                               "touchees": touch, "testees": n}))
